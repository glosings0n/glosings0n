/**
 * Practice 1: Automated Form Submission Email Receipt
 * Service: FormApp + GmailApp + SpreadsheetApp
 */

function handleWorkshopFormSubmission(e) {
  try {
    let studentName = 'Participant';
    let studentEmail = '';
    let selectedTrack = 'Google Workspace & Cloud Automation';

    // Handle execution both via Installable Event Trigger and Direct Testing
    if (e && e.namedValues) {
      // Triggered by Google Sheets form response link
      studentName = e.namedValues['Full Name'] ? e.namedValues['Full Name'][0] : 'Participant';
      studentEmail = e.namedValues['Email'] ? e.namedValues['Email'][0] : '';
      selectedTrack = e.namedValues['Track'] ? e.namedValues['Track'][0] : selectedTrack;
    } else {
      // Fallback for manual editor testing
      console.log('No event object detected. Using test participant parameters...');
      const props = PropertiesService.getScriptProperties();
      studentEmail = props.getProperty('ADMIN_EMAIL') || Session.getActiveUser().getEmail();
      studentName = 'Test Student (Editor Run)';
    }

    if (!studentEmail || !studentEmail.includes('@')) {
      console.warn(`Invalid or empty recipient email: "${studentEmail}". Aborting dispatch.`);
      return;
    }

    const emailSubject = `Confirmation: Registration Received | GDG Workspace Training 2026`;
    
    // Clean, responsive inline-styled HTML template (no external CSS dependencies)
    const htmlBody = `
      

        

          
GDG - Google Workspace Training

          
Registration Confirmation & Day V Roadmap

        

        

          
Hello ${studentName},

          
We are delighted to confirm that your workshop submission has been securely recorded in our system.

          
          

            
Session Summary:

            
    
              
  • Registered Track: ${selectedTrack}
    • 
              
  • Format: 5-Day Intensive Community Workshop
    • 
              
  • Automated Module: Day V - Google Apps Script Deep Dive
    • 
            

          


          

            Please ensure you have active access to script.google.com before our live hands-on session.
          

          

          

            Organized with GDG Afrique Francophone, GDG Pointe-Noire & GDG on Campus CUEA.

    `;

    const plainTextFallback = `Hello ${studentName},\n\nYour registration for GDG - Google Workspace Training has been confirmed!\nTrack: ${selectedTrack}\n\nSee you at the Day V Google Apps Script automation session!`;

      name: 'GDG Workspace Automation Lead'

  } catch (err) {
    console.error(`Failed to execute handleWorkshopFormSubmission: ${err.message}`);

}

function syncStudentsToGoogleContacts() {
  try {
    const props = PropertiesService.getScriptProperties();
    const spreadsheetId = props.getProperty('SPREADSHEET_ID');

    if (!spreadsheetId) {
      throw new Error('SPREADSHEET_ID not found in ScriptProperties. Run setupWorkshopEnvironment first.');
    }

    const spreadsheet = SpreadsheetApp.openById(spreadsheetId);
    const sheet = spreadsheet.getSheetByName('Attendance_Master');
    const data = sheet.getDataRange().getValues();

    if (data.length <= 1) {
      console.log('No student rows found in Attendance_Master.');
      return;
    }

    // 1. Locate or create the targeted Contact Group
    const groupName = 'GDG Workspace Training 2026';
    let contactGroup = ContactsApp.getContactGroup(groupName);

    if (!contactGroup) {
      contactGroup = ContactsApp.createContactGroup(groupName);
      console.log(`Created new Contact Group: "${groupName}"`);
    } else {
      console.log(`Using existing Contact Group: "${groupName}"`);
    }

    let createdCount = 0;

    const spreadsheet = SpreadsheetApp.openById(spreadsheetId);
    const sheet = spreadsheet.getSheetByName('Attendance_Master');
    const values = sheet.getDataRange().getValues();

    const students = [];
    let eligibleCount = 0;
    let totalScore = 0;

    for (let i = 1; i < values.length; i++) {
      const row = values[i];
      const rateNum = parseInt(String(row[9]).replace('%', ''), 10) || 0;
      const score = parseInt(row[10], 10) || 0;
      const status = row[11];
      const certUrl = row[12];

      if (rateNum >= 80) eligibleCount++;
      totalScore += score;

      students.push({
        id: row[0],
        name: row[1],
        email: row[2],
        track: row[3],
        rate: rateNum,
        score: score,
        status: status,
        certUrl: certUrl
      });
    }

    return {
      success: true,
      totalStudents: students.length,
      eligibleStudents: eligibleCount,
      averageScore: students.length ? Math.round(totalScore / students.length) : 0,
      students: students
    };
  } catch (err) {
    return { error: err.message };
  }
}

// 1. Programmatically install a nightly time-driven trigger
function installDailyAuditTrigger() {
  const functionName = 'auditStudentAttendanceAndNotify';
  
  // Clean up any existing trigger for the same function to avoid duplicate runs
  const currentTriggers = ScriptApp.getProjectTriggers();
  currentTriggers.forEach(trigger => {
    if (trigger.getHandlerFunction() === functionName) {
      ScriptApp.deleteTrigger(trigger);
      console.log(`Deleted pre-existing trigger for: ${functionName}`);
    }
  });

  // Schedule to execute daily at 18:00 (6:00 PM)
  ScriptApp.newTrigger(functionName)
    .timeBased()
    .everyDays(1)
    .atHour(18)
    .create();

  console.log(`=== AUTOMATIC TRIGGER INSTALLED ===`);
  console.log(`Function "${functionName}" will now run automatically every day at 18:00.`);
}

Production Quotas, Security Scopes & Best Practices

1. Production Invariants & Architectural Best Practices

Deploying reliable automations at scale requires adhering to core engineering constraints inherent to the Google Cloud Apps Script infrastructure:

  1. Atomic Batch Operations: Never invoke getValue() or setValue() inside a loop. Reading or writing cell-by-cell introduces significant network latency across the wire. Always use getValues() to read a 2D array into memory and setValues() to flush changes in a single atomic network trip (100x to 1000x faster).
  2. ScriptProperties for Environment Isolation: Never hardcode Folder IDs, Sheet IDs, or template keys directly in your business logic. Store configuration keys in PropertiesService.getScriptProperties() to keep code portable across testing and production sandboxes.
  3. Defensive Exception Boundaries: Always wrap external service calls (UrlFetchApp, GmailApp, DriveApp) in try { ... } catch (err) { console.error(...) } blocks. When interacting with external APIs, implement exponential backoff retry algorithms to handle transient network hiccups.
  4. Explicit Least-Privilege OAuth Scopes: In appsscript.json, explicitly declare only the permissions your application requires rather than accepting wide root access.

1. Invariants de Production & Bonnes Pratiques Architecturales

Le déploiement d'automatisations fiables à grande échelle exige le respect rigoureux des contraintes d'ingénierie de l'infrastructure Google Apps Script :

  1. Opérations par Lots Atomiques : N'invoquez jamais getValue() ou setValue() à l'intérieur d'une boucle. Lire ou écrire cellule par cellule multiplie les allers-retours réseau. Utilisez systématiquement getValues() pour charger la matrice en mémoire et setValues() pour enregistrer les modifications en un seul appel (gain de 100x à 1000x).
  2. ScriptProperties pour l'Isolation des Environnements : Ne codez jamais en dur les identifiants de dossiers, de feuilles ou de gabarits. Stockez ces variables dans PropertiesService.getScriptProperties() pour garantir la portabilité entre vos espaces de test et de production.
  3. Gestion Défensive des Erreurs : Encadrez systématiquement les appels aux services externes (UrlFetchApp, GmailApp, DriveApp) dans des blocs try { ... } catch (err) { console.error(...) }. Pour les APIs distantes, prévoyez un mécanisme de réessai avec backoff exponentiel.
  4. Principe du Moindre Privilège pour les Scopes OAuth : Déclarez explicitement dans le manifeste appsscript.json uniquement les périmètres d'accès indispensables.
